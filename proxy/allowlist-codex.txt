#  The API itself: every request the agent makes to the model.
^api\.openai\.com$

#  Sign-in and token refresh. Codex signs in through ChatGPT, so both of these
#  are part of authentication, not optional extras.
#
#  Expect the same trap as claude's platform.claude.com: leave these out and
#  the session starts fine, then fails when a token needs renewing, in a way
#  that looks like a proxy fault rather than an expired login.
^auth\.openai\.com$
^chatgpt\.com$
